name: Scarica foto
on:
  workflow_dispatch:
permissions:
  contents: write
jobs:
  foto:
    runs-on: ubuntu-latest
    timeout-minutes: 300
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - run: pip install pillow
      - name: Foto delle piante
        run: python strumenti/scarica_foto.py strumenti/piante.json
      - name: Foto delle ricette
        run: python strumenti/scarica_foto.py strumenti/ricette.json
      - name: Pagina dei crediti
        run: python strumenti/scarica_foto.py crediti
      - name: Salva nella repo
        run: |
          git config user.name "foto-bot"
          git config user.email "foto-bot@users.noreply.github.com"
          git add img strumenti/crediti.json crediti.html
          git commit -m "Foto scaricate da Wikimedia con crediti" || echo "niente di nuovo"
          git push
